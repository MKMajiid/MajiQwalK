cmake_minimum_required(VERSION 3.20)
project(MajiQwalK VERSION 0.1.0 LANGUAGES CXX)
option(MAJIQWALK_BUILD_PYTHON "Build Python extension" ON)
option(MAJIQWALK_OPENMP "Enable optional OpenMP coin kernel" ON)
option(MAJIQWALK_BUILD_TESTS "Build C++ tests" OFF)

add_library(majiqwalk_core STATIC src/core/engine.cpp)
target_include_directories(majiqwalk_core PUBLIC include)
target_compile_features(majiqwalk_core PUBLIC cxx_std_20)
set_target_properties(majiqwalk_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(majiqwalk_core PRIVATE /W4)
else()
  target_compile_options(majiqwalk_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MAJIQWALK_OPENMP)
  find_package(OpenMP QUIET)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(majiqwalk_core PUBLIC OpenMP::OpenMP_CXX)
  endif()
endif()

if(MAJIQWALK_BUILD_PYTHON)
  set(PYBIND11_FINDPYTHON ON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core src/bindings/module.cpp)
  target_link_libraries(_core PRIVATE majiqwalk_core)
  install(TARGETS _core LIBRARY DESTINATION majiqwalk RUNTIME DESTINATION majiqwalk)
endif()
if(MAJIQWALK_BUILD_TESTS)
  enable_testing()
  add_executable(test_core tests/cpp/test_core.cpp)
  target_link_libraries(test_core PRIVATE majiqwalk_core)
  add_test(NAME core COMMAND test_core)
endif()
